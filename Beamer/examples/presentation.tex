\begin{frame}{What Is LaTeX?}
    \begin{itemize}
        \item A document preparation system built on TeX.
        \item Uses commands to organise and format content.
        \item Useful for technical reports and research papers.
        \item Supports equations, tables, and cross-references.
    \end{itemize}

    \begin{block}{Document Workflow}
        Write the source, compile it, and inspect the PDF.
    \end{block}
\end{frame}
